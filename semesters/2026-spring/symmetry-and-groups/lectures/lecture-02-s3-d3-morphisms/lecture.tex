\begin{center}
{\Large \underline{Generators and rank of a group}}
\end{center}

\underline{Def. of generators}

A subset $S\subseteq G$ is called a set of generators of $G$
if every element of $G$ can be written as a finite product of
elements of $S$ and their inverses

\[
\boxed{G=\langle S\rangle}
\]

\textit{Ex}\quad Cyclic group $C_3$
\[
C_3=\left\{e,g,g^2\right\},\qquad g^3=e
\]
\[
\langle g\rangle=C_3
\]
\[
\operatorname{rank}(C_3)=1
\]

\textit{Ex}\quad Dihedral $D_3=\left\{e,r,r^{-1},\sig{1},\sig{2},\sig{3}\right\}$
\[
\sigma=\sig{1}
\]
\[
\sig{2}=r\sig{1}=r\sigma
\]
\[
\sig{3}=r^2\sigma=r^{-1}\sigma
\]
\[
D_3=\langle r,\sigma\rangle
\qquad \operatorname{rank}(D_3)=2
\]

\begin{center}
{\Large \underline{Group morphism}}
\end{center}

\underline{Homomorphism}

\[
\begin{array}{ll}
G&\text{(example, law }\circ\text{)}\\
H&\text{(comp. law }\diamond\text{)}
\end{array}
\]

A homomorphism is a mapping
\[
f^\circ:G\longrightarrow H,\qquad g\longmapsto h=f(g)
\]
Such that $g_1\circ g_2=g_3\subset G$
\[
\text{Then}\qquad f(g_1)\diamond f(g_2)=f(g_3)\subset H
\]

\[
D_3=\left\{e,r,r^{-1},\sig{1},\sig{2},\sig{3}\right\}
\qquad
C_2=\left\{e,c\right\},\quad c^2=e
\]
\[
f^\circ:D_3\longrightarrow C_2
\]
\[
e,r,r^{-1}\longrightarrow e
\qquad
\sig{1},\sig{2},\sig{3}\longrightarrow c
\]
\[
\circledast\quad \sig{1}\circ\sig{2}=r
\qquad\Longrightarrow\qquad
f(\sig{1}\circ\sig{2})=f(r)=e\subset C_2
\]
\[
f(\sig{1})\diamond f(\sig{2})=c\diamond c=e\subset C_2
\]

\[
\circledast\quad r\circ\sig{1}=\sig{2}
\]
\[
f(r\circ\sig{1})=f(\sig{2})=c\qquad\checkmark
\]
\[
f(r)\diamond f(\sig{1})=e\diamond c=c
\]

\underline{Counter example}
\[
\left\{
\begin{array}{rcl}
e\circ\sig{3}\circ r^{-1}&\longrightarrow&e\\
\sig{1}\circ\sig{2}\circ r^{-1}&\longrightarrow&c
\end{array}
\right.
\]
\[
\sig{3}\circ r^{-1}=\sig{2}
\]
\[
f(\sig{3}\circ r^{-1})=f(\sig{2})=c
\qquad\times
\]
\[
f(\sig{3})\diamond f(r^{-1})=e\diamond e=e
\]

\underline{Kernel}

Kernel of a homomorphism $f$ measures \textquotedblleft how much\textquotedblright\ of the
structure of $G$ collapses on the identity of the map $f$.
\[
\operatorname{Ker}(D_3)=\left\{e,r,r^{-1}\right\}
\]

\underline{Isomorphism}

\[
\longrightarrow\ \text{bijective homomorphism}\quad
G\cong H
\]

For finite group isomorphic $\equiv$ identical multiplication table
\[
C_4=\left\{e,g,g^2,g^3\right\},\qquad g^4=e
\]
\[
G_1=\left\{1,i,-1,-i\right\}
\]
\[
G_3=\left\{I,i\sigma_2,-I,-i\sigma_2\right\},
\qquad
\sigma_2=\left(\begin{array}{cc}0&-i\\i&0\end{array}\right)
\]
\[
\Longrightarrow\qquad C_4\cong G_1\cong G_3
\]

\underline{Endomorphism}
\[
\longrightarrow\ \text{homomorphism of a group }G\text{ on itself}
\qquad f^\circ:G\longrightarrow G
\]
\[
\text{Ex:}\quad f^\circ:\mathbb{R}\longrightarrow\mathbb{R},\quad f(x)=2x
\qquad\text{(with addition)}
\]

\vspace{0.8em}
\underline{Automorphism}
\[
\longrightarrow\ \text{bijective endomorphism}
\]

The set of all automorphism of a group $G$ is denoted
\[
\operatorname{Aut}(G)=\left\{f:G\longrightarrow G\mid f\ \text{bijective and }f(gh)=f(g)f(h)\right\}
\]

\textit{Ex}\quad group $(\mathbb{Z},+)$\quad $\longrightarrow\ \text{two automorphism}$
\[
f(n)=n
\qquad
f(n)=-n
\]
\[
\text{Therefore}\qquad \operatorname{Aut}(\mathbb{Z},+)\cong C_2
\]

\vspace{0.8em}
\underline{Cayley's Theorem}

Every finite group of order $n$ is isomorphic to a subgroup of the
symmetric group $S_n$
